Integration Example

// Voice Mode
import { HubUI } from '@hubui/client';

const session = await HubUI.connect({
  agentId: 'your-agent-id',
  apiKey: 'pk_live_xxxxx',
  mode: 'voice',
  userName: 'Demo User',
});

session.on('transcript', (text, speaker, isFinal) => {
  if (isFinal) console.log(`${speaker}: ${text}`);
});

// Text Mode
const textSession = await HubUI.connect({
  agentId: 'your-agent-id',
  apiKey: 'pk_live_xxxxx',
  mode: 'text',
});

textSession.on('message', (text) => console.log('Agent:', text));
await textSession.send('Hello!');
